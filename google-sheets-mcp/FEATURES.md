```typescript
// Create formatted financial report with charts
await Promise.all([
  createSheet({title: "P&L Dashboard", rowCount: 500}),
  writeRange({range: "A1:F20", values: financialData}),
  formatCells({range: "A1:F1", bold: true, backgroundColor: {red: 0.2, green: 0.4, blue: 0.8}}),
  createAdvancedChart({
    title: "Revenue Trends",
    chartType: "LINE",
    dataRange: "A1:B12",
    titleStyle: {fontSize: 16, bold: true}
  }),
  addConditionalFormatting({
    ranges: ["F2:F20"],
    conditionType: "NUMBER_LESS",
    values: [0],
    textColor: {red: 1, green: 0, blue: 0}
  })
]);
```

### **2. Data Analysis Workflow**
```typescript
// Complete data analysis pipeline
await Promise.all([
  addDataValidation({range: "A1:A100", validationType: "ONE_OF_LIST", values: ["Product A", "Product B", "Product C"]}),
  sortRange({range: "A1:D100", sortSpecs: [{dimensionIndex: 3, sortOrder: "DESCENDING"}]}),
  createPivotTable({
    sourceRange: "A1:D100",
    destinationSheetId: 1,
    rows: [{sourceColumnOffset: 0, showTotals: true}],
    values: [{sourceColumnOffset: 3, summarizeFunction: "SUM"}]
  }),
  createDataAnalysis({
    analysisType: "DESCRIPTIVE_STATS",
    dataRange: "D1:D100",
    outputRange: "F1:F10"
  })
]);
```

### **3. Interactive Dashboard**
```typescript
// Create interactive business dashboard
await batchUpdate({
  operations: [
    {type: "formatCells", data: {range: "A1:Z1", format: headerStyle}},
    {type: "updateCells", data: {range: "A1:J50", rows: dashboardData}},
    {type: "mergeCells", data: {range: "A1:J1"}}
  ]
});

await Promise.all([
  addSparklines({range: "K2:K20", dataRange: "C2:I2", chartType: "LINE"}),
  createAdvancedChart({
    chartType: "COMBO",
    series: [
      {type: "COLUMN", color: {red: 0.2, green: 0.6, blue: 0.9}},
      {type: "LINE", color: {red: 0.9, green: 0.2, blue: 0.2}, targetAxis: "RIGHT_AXIS"}
    ]
  }),
  protectRange({range: "A1:J1", description: "Header Protection", warningOnly: true})
]);
```
